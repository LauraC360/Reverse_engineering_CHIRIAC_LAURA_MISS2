DISSSASM INTERPRETATION

P4:

push ebp                             saves old ebp
mov ebp, esp                         base of stack frame 
sub    esp, 0x48                     esp = esp - 0x48
push ebx
push esi
push edi
mov    ecx, 0x4050a2
call   sub_0x00000040A                function call
mov    dword ptr [ebp - 8], 1         [ebp-8] = a = 1
mov    dword ptr [ebp - 4], 2         [ebp-4] = b = 2
mov    eax, dword ptr [ebp - 4]       eax = 2 (4 bytes)
sub    eax, 2                         eax = eax - 2 = 0
imul   ecx, dword ptr [ebp - 8], 3    ecx = [ebp-8] * 3 = a*3 = 3 (4 bytes)
add    ecx, 1                         ecx = ecx+1 = [ebp-8] * 3 + 1 = 4
imul   eax, ecx                       eax = eax * ecx = 0 * 4 = 0
mov    edx, dword ptr [ebp - 8]       edx = a = 1
sub    edx, 2                         edx = edx-2 = 1 - 2 = -1
imul   ecx, edx, 3                    ecx = edx * 3 = -3
sub    eax, ecx                       eax = eax - ecx = 0 - (-3) = 3
mov    dword ptr [ebp - 4], eax       b = eax = 3
xor    eax, eax                       eax = 0
pop    edi                            pop saved values
pop    esi
pop    ebx
mov    esp, ebp                       restore stack position
pop    ebp                            restore previous ebp
ret				      return 0;
